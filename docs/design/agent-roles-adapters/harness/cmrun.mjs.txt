import * as C from './cm.mjs'; import fs from 'node:fs'; import path from 'node:path';
const cmd=process.argv[2];
const tm=C.tmuxUp();
if(cmd==='up'){C.setup();C.mockUp();const pid=C.serverUp(tm);console.log('server pid',pid,'tmux',tm);
  for(let i=0;i<90;i++){await C.sleep(1000);try{const r=await fetch(`http://127.0.0.1:${C.CMP}/api/repositories`);if(r.ok){console.log('ready',i,JSON.stringify(await r.json()).slice(0,200));break;}}catch{}}
  process.exit(0);}
if(cmd==='restart'){const {execSync}=await import('node:child_process');const pids=execSync(`lsof -nP -iTCP:${C.CMP} -sTCP:LISTEN -t || true`,{encoding:'utf8'}).trim().split('\n').filter(Boolean);const sp=fs.readFileSync(C.ROOT+'/server.pid','utf8').trim();for(const p of [...pids,sp]){try{process.kill(+p,'SIGTERM');}catch{}}await C.sleep(5000);const pid=C.serverUp(tm);for(let i=0;i<90;i++){await C.sleep(1000);try{const r=await fetch(`http://127.0.0.1:${C.CMP}/api/repositories`);if(r.ok){console.log('restarted',pid,'after',i);break;}}catch{}}process.exit(0);}
const WTS=async()=>{const j=await C.api('/api/worktrees');return j.worktrees??j;};
if(cmd==='ls'){await C.api('/api/repositories/sync',{method:'POST'}).catch(()=>null);const w=await WTS();console.log(w.map((x)=>x.id+' '+path.basename(x.path)).join('\n'));process.exit(0);}
const tool=process.argv[3];const kind=process.argv[4]??'role';
const w=(await WTS()).find((x)=>path.basename(x.path)===`${tool}-${kind}`);const id=w.id;
const CASE=`CASE${tool.replace(/-/g,'')}${kind}`;
const rec=JSON.parse(fs.existsSync(`${C.ROOT}/rec-${tool}-${kind}.json`)?fs.readFileSync(`${C.ROOT}/rec-${tool}-${kind}.json`,'utf8'):'{"steps":{}}');rec.tool=tool;rec.kind=kind;rec.worktree=id;
const save=()=>fs.writeFileSync(`${C.ROOT}/rec-${tool}-${kind}.json`,JSON.stringify(rec,null,1));
const inst=['--instance',tool];
const mark=(since,word)=>{const rs=C.reads(since).filter((r)=>r.method==='POST'&&JSON.stringify(r.body).includes(word)&&(r.body.tools??[]).length>0);const b=rs.pop()?.body;return b?{requests:rs.length+1,role:C.where(b,'ROLE_MARKER_K3'),repo:C.where(b,'REPO_MARKER_7Q'),tools:(b.tools??[]).length,toolResult:/VERIFY_(OK|NG)/.test(JSON.stringify(b))?(/VERIFY_OK/.test(JSON.stringify(b))?'VERIFY_OK':'VERIFY_NG'):null}:null;};
async function approvalLoop(args,limit=5){const seq=[];for(let i=0;i<limit;i++){const r=C.cli(tm,'wait',id,...inst,'--timeout','70',...args);seq.push({code:r.code,out:r.out.slice(0,420),err:r.err.slice(-200)});if(r.code===10){const a=C.cli(tm,'respond',id,'--default',...inst);seq.push({respond:a.code,out:(a.out+a.err).slice(0,200)});if(a.code!==0){const b=C.cli(tm,'respond',id,'1',...inst);seq.push({respond1:b.code,out:(b.out+b.err).slice(0,200)});}await C.sleep(2000);continue;}break;}return seq;}
if(cmd==='send1'){const since=Date.now();const r=C.cli(tm,'send',id,`${CASE} turn1 hello`,'--agent',tool,...inst);rec.steps.send1={code:r.code,out:r.out.slice(0,300),err:r.err.slice(-500)};const wr=C.cli(tm,'wait',id,...inst,'--timeout','60');rec.steps.wait1={code:wr.code,out:wr.out.slice(0,420),err:wr.err.slice(-300)};await C.sleep(800);rec.steps.mock1=mark(since,`${CASE} turn1`);console.log(JSON.stringify(rec.steps,null,0).slice(0,1100));save();}
if(cmd==='fix'){const since=Date.now();const msg=`${CASE} turn2 CMRUN<<sed -i.bak s/1+2/1+1/ calc.sh; rm -f calc.sh.bak; sh verify.sh; echo RC=$?>>`;const r=C.cli(tm,'send',id,msg,...inst);rec.steps.send2={code:r.code,err:r.err.slice(-300)};rec.steps.approval=await approvalLoop(['--on-prompt','agent','--verify']);await C.sleep(800);rec.steps.mock2=mark(since,`${CASE} turn2`);rec.steps.fileAfter=fs.readFileSync(w.path+'/calc.sh','utf8');console.log(JSON.stringify({send2:rec.steps.send2,approval:rec.steps.approval,mock2:rec.steps.mock2,f:rec.steps.fileAfter}).slice(0,1800));save();}
if(cmd==='third'){const since=Date.now();const r=C.cli(tm,'send',id,`${CASE} turn3 afterrestart`,...inst);rec.steps.send3={code:r.code,err:r.err.slice(-300)};const wr=C.cli(tm,'wait',id,...inst,'--timeout','60');rec.steps.wait3={code:wr.code,out:wr.out.slice(0,420)};await C.sleep(800);rec.steps.mock3=mark(since,`${CASE} turn3`);console.log(JSON.stringify({s:rec.steps.send3,w:rec.steps.wait3,m:rec.steps.mock3}).slice(0,900));save();}
if(cmd==='contract'){const since=Date.now();const fp=`${w.path}/.commandmate/tasks/t.yaml`;fs.mkdirSync(path.dirname(fp),{recursive:true});fs.writeFileSync(fp,`version: 1\ntitle: "probe"\ngoal: |\n  ${CASE} turn4 CMRUN<<echo done > note.txt; sh verify.sh; echo RC=$?>>\nscope:\n  allow:\n    - note.txt\n    - calc.sh\n  deny: []\nverify:\n  gates:\n    - check\nsuccess:\n  requireWorkEvidence: true\n  requireScopeClean: true\n`);const r=C.cli(tm,'send',id,'--contract','.commandmate/tasks/t.yaml',...inst);rec.steps.sendC={code:r.code,out:r.out.slice(0,300),err:r.err.slice(-400)};rec.steps.approvalC=await approvalLoop(['--on-prompt','agent','--verify']);await C.sleep(800);rec.steps.mockC=mark(since,`${CASE} turn4`);console.log(JSON.stringify({s:rec.steps.sendC,a:rec.steps.approvalC,m:rec.steps.mockC}).slice(0,1800));save();}
if(cmd==='pane'){const sess=C.sessions().filter((s)=>s.includes(id));console.log(sess);for(const s of sess)console.log(C.pane(s).replace(/\n\s*\n+/g,'\n').slice(-1200));}
process.exit(0);
