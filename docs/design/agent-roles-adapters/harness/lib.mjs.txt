export const flat = (b) => JSON.stringify(b);
// 指示層（system/developer）の本文を取り出す
export function layers(body) {
  const out = { system: '', developer: '', user: '' };
  const t = (c) => typeof c === 'string' ? c : Array.isArray(c) ? c.map((p) => p.text ?? p.input_text ?? '').join('\n') : '';
  if (body.system) out.system += t(body.system);
  if (body.instructions) out.system += t(body.instructions);
  for (const m of [...(body.messages ?? []), ...(Array.isArray(body.input) ? body.input : [])]) {
    if (m.role === 'system') out.system += '\n' + t(m.content);
    else if (m.role === 'developer') out.developer += '\n' + t(m.content);
    else if (m.role === 'user') out.user += '\n' + t(m.content);
  }
  return out;
}
export const toolNames = (b) => (b.tools ?? []).map((t) => t.name ?? t.function?.name ?? t.type).sort();
export function whereIs(body, marker) {
  const l = layers(body); const w = [];
  for (const k of ['system', 'developer', 'user']) if (l[k].includes(marker)) w.push(k);
  return w;
}

// セッションを起動: launch.sh を書いて new-session
export function startPane(name, repo, env, argv, { width = 200, height = 60, preamble = '' } = {}) {
  const dir = path.join(ROOT, 'launch'); fs.mkdirSync(dir, { recursive: true });
  const f = path.join(dir, name + '.sh');
  const q = (s) => `'${String(s).replace(/'/g, `'\\''`)}'`;
  const envs = Object.entries({ PATH: PATHV, TERM: 'xterm-256color', LANG: 'en_US.UTF-8', ...env }).map(([k, v]) => `${k}=${q(v)}`).join(' ');
  fs.writeFileSync(f, `#!/bin/bash\ncd ${q(repo)}\n${preamble}\nenv -i ${envs} ${argv.map(q).join(' ')}\necho "__EXIT=$?"\nexec sleep 86400\n`);
  try { tmux('kill-session', '-t', `=${name}`); } catch {}
  tmux('new-session', '-d', '-s', name, '-x', String(width), '-y', String(height), `bash ${q(f)}`);
  return { name, file: f };
}

export const ROLE_TEXT = 'ROLE_MARKER_K3: You are a test role. End every answer with ROLESTAMP.';
export function applyRole(tool, d, h, repo, variant, body = ROLE_TEXT, label = 'x') {
  const w = (p, s) => { fs.mkdirSync(path.dirname(p), { recursive: true }); fs.writeFileSync(p, s, { mode: 0o600 }); };
  const rf = `${h}/.cmrole/role-${label}.md`;
  const out = { env: { ...d.env }, cmd: [...d.cmd], artifacts: [] };
  const art = (p, s) => { w(p, s); out.artifacts.push({ path: p.replace(h, '$HOME'), content: s.length > 2000 ? s.slice(0, 2000) + '…' : s }); };
  if (variant === 'none') return out;
  const toml = (s) => JSON.stringify(s);
  switch (tool) {
    case 'claude': art(rf, body); out.cmd.push('--append-system-prompt-file', rf); break;
    case 'codex': art(`${h}/.codex/cm-role-${label}.config.toml`, `developer_instructions = ${toml(body)}\n`); out.cmd.push('-p', `cm-role-${label}`); break;
    case 'command-code': { const f = `${h}/.cmrole/cm-role-${label}.ts`; art(f, `export default function (cmd) {\n  cmd.hooks({ appendSystemPrompt: () => ${JSON.stringify(body)} });\n}\n`); out.cmd.push('--mod', f); break; }
    case 'opencode': art(rf, body); out.env.OPENCODE_CONFIG_CONTENT = JSON.stringify({ instructions: [rf] }); break;
    case 'opencode-v2':
      if (variant === 'instructions-content') { art(rf, body); out.env.OPENCODE_CONFIG_CONTENT = JSON.stringify({ instructions: [rf] }); }
      else if (variant === 'instructions-dir') { art(rf, body); const dir = `${h}/.cmrole/cfg-${label}`; art(`${dir}/opencode.jsonc`, JSON.stringify({ instructions: [rf] })); out.env.OPENCODE_CONFIG_DIR = dir; }
      else if (variant === 'agent-md' ) { const dir = `${h}/.cmrole/cfg-${label}`; art(`${dir}/agents/cm-role-${label}.md`, `---\ndescription: CommandMate role\nmode: primary\nhidden: true\n---\n${body}\n`); art(`${dir}/opencode.jsonc`, JSON.stringify({ default_agent: `cm-role-${label}` })); out.env.OPENCODE_CONFIG_DIR = dir; }
      else if (variant === 'overlay') { art(rf, body); out.env.OPENCODE_CONFIG_CONTENT = JSON.stringify({ agents: { build: { system: `{file:${rf}}` } } }); }
      else if (variant === 'agent-inline') { out.env.OPENCODE_CONFIG_CONTENT = JSON.stringify({ default_agent: `cm-role-${label}`, agents: { [`cm-role-${label}`]: { description: 'role', mode: 'primary', system: body } } }); }
      else throw new Error(variant);
      break;
    case 'copilot': art(`${h}/.copilot/agents/cm-role-${label}.agent.md`, `---\nname: cm-role-${label}\ndescription: CommandMate role\ninfer: false\n---\n${body}\n`); out.cmd.push('--agent', `cm-role-${label}`); break;
  }
  return out;
}

const norm = (s, base) => s.split(base).join('<D>').replace(/\d{4}-\d{2}-\d{2}/g, '<DATE>').replace(/[0-9a-f]{8}-[0-9a-f-]{27}/g, '<UUID>');
export function metrics(body, base) {
  const l = layers(body);
  const sys = norm(l.system + '\n' + l.developer, base);
  return { url: null, model: body.model, tools: toolNames(body), toolsHash: sha(toolNames(body).join(',')), systemLen: l.system.length, developerLen: l.developer.length, systemHash: sha(sys), roleIn: whereIs(body, 'ROLE_MARKER_K3'), repoIn: whereIs(body, 'REPO_MARKER_7Q'), sysLines: [...new Set(sys.split('\n').map((x) => x.trim()).filter((x) => x.length > 20))] };
}
